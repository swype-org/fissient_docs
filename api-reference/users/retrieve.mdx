---
title: "Retrieve a user"
api: "GET /v1/users/{id}"
description: "Returns a user and their current guardrails."
---

## Path parameters

<ParamField path="id" type="string" required>
  The user's id, such as `usr_01M761XYYTGZK43AP9XK0WKQ97`.
</ParamField>

## The user object

<ResponseField name="id" type="string">
  Unique id, prefixed `usr_`.
</ResponseField>
<ResponseField name="object" type="string">
  Always `user`.
</ResponseField>
<ResponseField name="partner_user_ref" type="string">
  Your own id for this customer.
</ResponseField>
<ResponseField name="residence_country" type="string">
  Two-letter country code.
</ResponseField>
<ResponseField name="nationality" type="string">
  Two-letter country code.
</ResponseField>
<ResponseField name="investor_type" type="string">
  `retail` or `professional`.
</ResponseField>
<ResponseField name="display_currency" type="string">
  Currency used for display values.
</ResponseField>
<ResponseField name="kyc_attested" type="boolean">
  Whether you've attested the user's identity.
</ResponseField>
<ResponseField name="status" type="string">
  `active` for a user in good standing.
</ResponseField>
<ResponseField name="livemode" type="boolean">
  `true` for live users, `false` in the sandbox.
</ResponseField>
<ResponseField name="policy" type="object">
  The user's [guardrails](/concepts/guardrails). Empty when none are set.
</ResponseField>
<ResponseField name="linked_home_wallet" type="object | null">
  The user's [linked wallet](/api-reference/users/link-wallet), if any.
</ResponseField>
<ResponseField name="deposit_instructions" type="object">
  The user's deposit addresses. Prefer [deposit instructions](/api-reference/funding/deposit-instructions), which lists them by network.
</ResponseField>
<ResponseField name="created_at" type="string">
  When the user was created.
</ResponseField>

<RequestExample>
  ```bash cURL
  curl https://invest-api.blink.cash/v1/users/usr_01M761XYYTGZK43AP9XK0WKQ97 \
    -H "Authorization: Bearer $FISSIENT_KEY"
  ```
</RequestExample>
