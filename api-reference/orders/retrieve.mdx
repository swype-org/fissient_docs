---
title: "Retrieve an order"
api: "GET /v1/orders/{id}"
description: "Returns an order and its current status."
---

## Path parameters

<ParamField path="id" type="string" required>
  The order's id, such as `ord_01M761Y4TV5PH23YVNCTDW63W1`.
</ParamField>

## Query parameters

<ParamField query="wait" type="integer">
  Seconds to hold the response, up to 60, until the order completes, fails, or needs approval or funding. Use it instead of a polling loop.
</ParamField>

## The order object

<ResponseField name="id" type="string">
  Unique id, prefixed `ord_`.
</ResponseField>
<ResponseField name="object" type="string">
  Always `order`.
</ResponseField>
<ResponseField name="quote_id" type="string | null">
  The quote this order executed. `null` for limit and stop orders.
</ResponseField>
<ResponseField name="user_id" type="string">
  The user the order belongs to.
</ResponseField>
<ResponseField name="status" type="string">
  One of `open`, `awaiting_funding`, `needs_approval`, `routing`, `settling`, `needs_reconciliation`, `completed`, `failed`, `canceled` or `expired`. See [order statuses](/concepts/quotes-and-orders#order-statuses).
</ResponseField>
<ResponseField name="order_type" type="string">
  `market`, `limit`, `stop`, `stop_limit` or `take_profit`.
</ResponseField>
<ResponseField name="side" type="string">
  `buy` or `sell`.
</ResponseField>
<ResponseField name="instrument" type="string">
  The instrument id.
</ResponseField>
<ResponseField name="limit_price_usd" type="string | null">
  The limit price, for limit and stop-limit orders.
</ResponseField>
<ResponseField name="trigger_price_usd" type="string | null">
  The trigger price, for stop, stop-limit and take-profit orders.
</ResponseField>
<ResponseField name="time_in_force" type="string | null">
  `gtc` or `day`, for limit and stop orders.
</ResponseField>
<ResponseField name="expires_at" type="string | null">
  When an open order expires if it hasn't triggered.
</ResponseField>
<ResponseField name="triggered_at" type="string | null">
  When a limit or stop order's price was met.
</ResponseField>
<ResponseField name="leveraged" type="object | null">
  `null` for an ordinary trade. For a leveraged order: `direction` (`long` or `short`), `leverage`, and `action` (`open` or `close`).
</ResponseField>
<ResponseField name="input" type="object">
  What the user gives up: `asset`, `amount` and `value_usd`.
</ResponseField>
<ResponseField name="output" type="object | null">
  What the user received: `asset`, `units` and `value_usd`. `null` until the order completes.
</ResponseField>
<ResponseField name="effective_price_usd" type="string | null">
  The price per unit achieved, fees included. `null` until the order completes.
</ResponseField>
<ResponseField name="fees" type="object">
  `all_in_fee_usd` is the complete cost of the order.
</ResponseField>
<ResponseField name="funding_instructions" type="object | null">
  Set while the order is `awaiting_funding`: the `asset`, `amount`, `chain`, `from_address` and `to_address` of the transfer it's waiting for, and the `funding_id` of that transfer.
</ResponseField>
<ResponseField name="failure" type="object | null">
  Set when the order failed: a `code` and a `message`.
</ResponseField>
<ResponseField name="progress" type="object">
  `steps_total` and `steps_completed`. Useful for a progress indicator while an order is executing.
</ResponseField>
<ResponseField name="created_at" type="string">
  When the order was placed.
</ResponseField>
<ResponseField name="completed_at" type="string | null">
  When the order completed.
</ResponseField>
